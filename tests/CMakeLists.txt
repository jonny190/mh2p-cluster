cmake_minimum_required(VERSION 3.16)
project(mib2p_lifecycle_tests C)
enable_testing()
add_executable(lifecycle_native_tests lifecycle_native_tests.c)
target_include_directories(lifecycle_native_tests PRIVATE ../src)
if(MSVC)
  target_compile_options(lifecycle_native_tests PRIVATE /W4)
else()
  target_compile_options(lifecycle_native_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME lifecycle_native COMMAND lifecycle_native_tests)
add_executable(receiver_ring_tests receiver_ring_tests.c)
target_include_directories(receiver_ring_tests PRIVATE ../src)
add_test(NAME receiver_ring COMMAND receiver_ring_tests)
add_executable(lifecycle_native_peer lifecycle_native_peer.c)
target_include_directories(lifecycle_native_peer PRIVATE ../src)
if(MSVC)
  target_compile_definitions(lifecycle_native_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_definitions(lifecycle_native_peer PRIVATE _CRT_SECURE_NO_WARNINGS)
  add_library(lifecycle_daemon_syntax OBJECT lifecycle_daemon_compile.c)
  target_include_directories(lifecycle_daemon_syntax PRIVATE ../src)
  target_compile_definitions(lifecycle_daemon_syntax PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(lifecycle_daemon_syntax PRIVATE /W4 /WX)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  set(send_fixture "${CMAKE_CURRENT_BINARY_DIR}/generation_send.inc")
  add_custom_command(OUTPUT "${send_fixture}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/extract_generation_send.py"
      "${CMAKE_CURRENT_SOURCE_DIR}/../src/gal_cluster.c" "${send_fixture}"
    DEPENDS ../src/gal_cluster.c ../scripts/extract_generation_send.py)
  add_executable(generation_adapter_tests generation_adapter_tests.c "${send_fixture}")
  target_include_directories(generation_adapter_tests PRIVATE ../src "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(generation_adapter_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
  add_test(NAME generation_adapters COMMAND generation_adapter_tests)
  add_executable(ring_process_tests ring_process_tests.c)
  target_include_directories(ring_process_tests PRIVATE ../src)
  target_compile_definitions(ring_process_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
  add_test(NAME ring_process COMMAND ring_process_tests "${CMAKE_CURRENT_BINARY_DIR}/ring-process.bin")
  set_tests_properties(ring_process PROPERTIES TIMEOUT 45)
endif()
